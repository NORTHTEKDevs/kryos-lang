fn fib(n: i64) -> i64 {
    if n < 2 { n } else { fib(n - 1) + fib(n - 2) }
}

fn main() {
    println(fib(10))        // 55
    println(factorial(10))  // 3628800
    println(sum_to(100))    // 5050
